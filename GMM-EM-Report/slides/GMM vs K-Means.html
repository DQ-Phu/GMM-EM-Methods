def standardize(X):
    std = X.std(axis=0)
    std[std == 0] = 1
    return (X - X.mean(axis=0)) / std

def pca_fit(X, k=2):
    mu = X.mean(axis=0)
    vals, vecs = np.linalg.eigh(np.cov(X - mu, rowvar=False))
    order = np.argsort(vals)[::-1][:k]
    explained_var = vals[order].sum() / vals.sum()
    return mu, vecs[:, order], explained_var

def pca_transform(X, mu, W):
    return (X - mu) @ W

K-Means: Tối Ưu Hóa Tọa Độ Khối & K-Means++ Seeding

Hàm Mục Tiêu (WCSS / Inertia):
$$J = \sum_{i=1}^{n} \sum_{k=1}^{K} r_{ik} \|\mathbf{x}_i - \boldsymbol{\mu}_k\|^2, \quad r_{ik} \in \{0, 1\}$$

Gán cứng: $r_{ik} = 1$ khi $k = \arg\min_j \|\mathbf{x}_i - \boldsymbol{\mu}_j\|^2$. Cập nhật tâm: $\boldsymbol{\mu}_k = \frac{1}{|C_k|}\sum_{i \in C_k} \mathbf{x}_i$.

Kỹ thuật Vector hóa Khoảng cách Euclid Ma trận:

Tránh vòng lặp for kép lồng nhau thông qua khai triển hằng đẳng thức ma trận:

$$\|\mathbf{A} - \mathbf{B}\|^2 = \|\mathbf{a}\|^2 + \|\mathbf{b}\|^2 - 2\mathbf{a}\mathbf{b}^T$$

Seeding k-means++ chọn tâm tiếp theo tỷ lệ thuận với $D(\mathbf{x})^2$.

kmeans.py Matrix Vectorization 
def sq_dists(A, B):
    d = (A**2).sum(1)[:, None] + (B**2).sum(1)[None, :] - 2 * A @ B.T
    return np.maximum(d, 0)

def kmeans_once(X, k, rng, max_iter=300, tol=1e-4):
    centers = [X[rng.integers(len(X))]]
    for _ in range(k - 1):
        d2 = sq_dists(X, np.array(centers)).min(axis=1)
        centers.append(X[rng.choice(len(X), p=d2 / d2.sum())])
    C = np.array(centers)
    
    for _ in range(max_iter):
        labels = sq_dists(X, C).argmin(axis=1)
        newC = np.array([X[labels == j].mean(0) for j in range(k)])
        if ((newC - C)**2).sum() <= tol * X.var(0).mean():
            break
        C = newC
    inertia = sq_dists(X, C)[np.arange(len(X)), labels].sum()
    return C, labels, inertia

GMM-EM: Tránh Tràn Số & Nghịch Đảo Ma Trận Bằng Cholesky

Mật độ Gauss trong không gian Log:
$$\ln \mathcal{N}(\mathbf{x} \mid \boldsymbol{\mu}, \boldsymbol{\Sigma}) = -\frac{1}{2}\Big( D\ln(2\pi) + \ln|\boldsymbol{\Sigma}| + (\mathbf{x}-\boldsymbol{\mu})^T \boldsymbol{\Sigma}^{-1} (\mathbf{x}-\boldsymbol{\mu}) \Big)$$
Phân rã Cholesky $\boldsymbol{\Sigma} = \mathbf{L}\mathbf{L}^T$ giúp giải quyết triệt để 2 vấn đề lớn:
$$\ln|\boldsymbol{\Sigma}| = 2\sum_{i=1}^D \ln L_{ii}, \qquad (\mathbf{x}-\boldsymbol{\mu})^T \boldsymbol{\Sigma}^{-1} (\mathbf{x}-\boldsymbol{\mu}) = \|\mathbf{L}^{-1}(\mathbf{x}-\boldsymbol{\mu})\|^2$$
Kỹ thuật ổn định số học:

1. Cộng reg = 1e-6 * I vào mỗi ma trận hiệp phương sai để chống suy biến (Singularity).

2. Hàm logsumexp tính $\ln \sum \exp(a_k)$ bằng cách trừ đi $\max(a)$, bảo đảm không bao giờ bị tràn dưới (Underflow).

gmm_stable_math.py Cholesky solve 
def log_gauss(X, mu, cov):
    d = X.shape[1]
    L = np.linalg.cholesky(cov)
    z = np.linalg.solve(L, (X - mu).T)
    maha = (z ** 2).sum(0)
    log_det = 2.0 * np.log(np.diag(L)).sum()
    return -0.5 * (d * np.log(2 * np.pi) + maha) - 0.5 * log_det

def logsumexp(a, axis):
    m = a.max(axis=axis, keepdims=True)
    return (m + np.log(np.exp(a - m).sum(axis=axis, keepdims=True))).squeeze()

Slide 05 • Vòng lặp EM

Cập Nhật Đồng Thời E-Step & M-Step Với Tốc Độ Cao

1. E-Step: Responsibility $\mathbf{R}$
$$r_{ik} = \frac{\pi_k \mathcal{N}(\mathbf{x}_i \mid \boldsymbol{\mu}_k, \boldsymbol{\Sigma}_k)}{\sum_{j=1}^K \pi_j \mathcal{N}(\mathbf{x}_i \mid \boldsymbol{\mu}_j, \boldsymbol{\Sigma}_j)}$$
2. M-Step: Closed-form updates
$$N_k = \sum_{i=1}^n r_{ik}, \quad \pi_k = \frac{N_k}{n}, \quad \boldsymbol{\mu}_k = \frac{1}{N_k}\sum_{i=1}^n r_{ik}\mathbf{x}_i$$ $$\boldsymbol{\Sigma}_k = \frac{1}{N_k}\sum_{i=1}^n r_{ik}(\mathbf{x}_i - \boldsymbol{\mu}_k)(\mathbf{x}_i - \boldsymbol{\mu}_k)^T + \text{reg}\cdot\mathbf{I}$$
Kết quả hội tụ trên EV Dataset ($K=2$):
  • • Converged: True (sau 5 vòng lặp)
  • • Final Log-likelihood: -110020.33
  • • Mixing weights: [0.693, 0.307]
em_steps.py BLAS Vectorized 
def e_step(X, pi, mu, cov):
    logp = np.column_stack([
        np.log(pi[k]) + log_gauss(X, mu[k], cov[k]) for k in range(len(pi))
    ])
    ll = logsumexp(logp, axis=1)
    return np.exp(logp - ll[:, None]), ll.mean()

def m_step(X, R, reg=1e-6):
    n, d = X.shape
    Nk = R.sum(0) + 1e-10
    mu = (R.T @ X) / Nk[:, None]
    cov = np.empty((R.shape[1], d, d))
    for k in range(R.shape[1]):
        diff = X - mu[k]
        cov[k] = (R[:, k, None] * diff).T @ diff / Nk[k] + reg * np.eye(d)
    return Nk / n, mu, cov

Chứng Minh: K-Means Là Trường Hợp Giới Hạn Của GMM

Giả định mô hình GMM có ma trận hiệp phương sai hình cầu đẳng hướng chung $\boldsymbol{\Sigma}_k = \epsilon^2 \mathbf{I}$ và trọng số đều nhau $\pi_k = \frac{1}{K}$. Khi đó mật độ phân phối: $\mathcal{N}(\mathbf{x}_i \mid \boldsymbol{\mu}_k, \epsilon^2\mathbf{I}) \propto \exp\left(-\frac{\|\mathbf{x}_i - \boldsymbol{\mu}_k\|^2}{2\epsilon^2}\right)$.

1. Giới Hạn Bước E (Softmax Nhiệt Độ $T = 2\epsilon^2$)
$$\gamma_{ik} = \frac{\exp\left(-\frac{\|\mathbf{x}_i - \boldsymbol{\mu}_k\|^2}{2\epsilon^2}\right)}{\sum_{j=1}^K \exp\left(-\frac{\|\mathbf{x}_i - \boldsymbol{\mu}_j\|^2}{2\epsilon^2}\right)}$$

Khi cho phương sai tiến dần về $0$ ($\epsilon \to 0$), số hạng có khoảng cách Euclid nhỏ nhất áp đảo hoàn toàn mẫu số:

$$\lim_{\epsilon \to 0} \gamma_{ik} = \begin{cases} 1 & \text{nếu } k = \arg\min_j \|\mathbf{x}_i - \boldsymbol{\mu}_j\|^2 \\ 0 & \text{ngược lại} \end{cases} \quad \equiv r_{ik}$$

→ Gán mềm $\gamma_{ik}$ suy biến chính xác thành gán cứng $r_{ik}$ của K-means!

2. Giới Hạn Bước M (Cập Nhật Trọng Tâm)
$$\boldsymbol{\mu}_k^{(t+1)} = \frac{\sum_{i=1}^n \gamma_{ik} \mathbf{x}_i}{\sum_{i=1}^n \gamma_{ik}}$$

Khi $\epsilon \to 0$, vì $\gamma_{ik} \to r_{ik} \in \{0, 1\}$ và mẫu số $N_k \to |C_k|$, công thức tự động trở thành:

$$\lim_{\epsilon \to 0} \boldsymbol{\mu}_k^{(t+1)} = \frac{1}{|C_k|} \sum_{\mathbf{x}_i \in C_k} \mathbf{x}_i$$

→ Trực tiếp trùng khớp với phép cập nhật trọng tâm của K-means!

Cài Đặt Hàm Silhouette, Davies-Bouldin và ARI

1. Silhouette ($s_i \in [-1, 1]$): $$s_i = \frac{b_i - a_i}{\max(a_i, b_i)}$$ Xử lý khối chunks(size=1000) để tối ưu bộ nhớ RAM.
2. Davies-Bouldin Index ($\downarrow$): $$DB = \frac{1}{K}\sum_{i=1}^K \max_{j \ne i} \frac{S_i + S_j}{d(\mathbf{c}_i, \mathbf{c}_j)}$$
3. Adjusted Rand Index (ARI): $$ARI = \frac{\sum_{ij}\binom{n_{ij}}{2} - \text{Expected}}{\frac{1}{2}(\sum_i \binom{a_i}{2} + \sum_j \binom{b_j}{2}) - \text{Expected}}$$
metrics.py Memory Safe 
def adjusted_rand(a, b):
    _, a = np.unique(a, return_inverse=True)
    _, b = np.unique(b, return_inverse=True)
    cont = np.zeros((a.max() + 1, b.max() + 1))
    np.add.at(cont, (a, b), 1)
    c2 = lambda x: x * (x - 1) / 2
    sum_ij, sum_a, sum_b = c2(cont).sum(), c2(cont.sum(1)).sum(), c2(cont.sum(0)).sum()
    expected = sum_a * sum_b / c2(len(a))
    return (sum_ij - expected) / (0.5 * (sum_a + sum_b) - expected)

def davies_bouldin(X, labels):
    ids = np.unique(labels)
    cent = np.array([X[labels == c].mean(0) for c in ids])
    scat = np.array([np.linalg.norm(X[labels == c] - cent[i], axis=1).mean() for i, c in enumerate(ids)])
    sep = np.sqrt(sq_dists(cent, cent))
    np.fill_diagonal(sep, np.inf)
    return ((scat[:, None] + scat[None, :]) / sep).max(1).mean()

Chiến Lược Lai Tối Ưu: Khởi Tạo GMM-EM Bằng K-Means

Quy trình khởi tạo tham số $\boldsymbol{\theta}^{(0)}$:

Thay vì gán ngẫu nhiên ma trận $\mathbf{R}^{(0)}$, ta dùng kết quả nhãn cứng của K-means làm phân phối trách nhiệm ban đầu:

$$\mathbf{R}^{(0)} = \text{OneHot}(\text{labels}_{\text{kmeans}})$$

Sau đó chạy trực tiếp bước M đầu tiên để ước lượng ngay $\boldsymbol{\mu}_k^{(0)}, \boldsymbol{\Sigma}_k^{(0)}, \pi_k^{(0)}$.

⚡ Minh Chứng Thực Nghiệm:

Nhờ khởi tạo từ kmeans_once, thuật toán GMM-EM trên tập dữ liệu $15.000$ mẫu đạt trạng thái hội tụ chuẩn chỉ sau đúng 5 vòng lặp (Iterations: 5) với Log-likelihood: -110020.33!

gmm_warm_start.py Warm Start 
def gmm_once(X, k, rng, max_iter=200, tol=1e-3):
    # 1. Khởi tạo nhãn từ K-means
    _, labels, _ = kmeans_once(X, k, rng)
    
    # 2. Khởi tạo tham số ban đầu qua bước M trên nhãn One-hot
    pi, mu, cov = m_step(X, np.eye(k)[labels])
    
    prev, converged = -np.inf, False
    for it in range(1, max_iter + 1):
        R, ll = e_step(X, pi, mu, cov)
        pi, mu, cov = m_step(X, R)
        if abs(ll - prev) < tol:
            converged = True
            break
        prev = ll
    return dict(pi=pi, mu=mu, cov=cov, R=R, ll=ll, n_iter=it)

Khi Nào Dùng GMM, Khi Nào Dùng K-Means?

1. Ưu Tiên K-Means
  • • Dữ liệu rất lớn ($N \ge 10^5$): Cần tối ưu tốc độ xử lý nhanh và tiết kiệm bộ nhớ RAM.
  • • Cụm hình cầu: Các biến độc lập, phương sai đồng đều sau khi chuẩn hóa.
  • • Phân vùng thô: Đóng vai trò làm bước tiền xử lý cho các tác vụ hạ nguồn.
Độ phức tạp: Tuyến tính $\mathcal{O}(NKD)$
2. Ưu Tiên GMM-EM
  • • Cụm elipxoit: Tồn tại tương quan mạnh giữa các biến (thu nhập vs. quãng đường).
  • • Đo độ bất định: Cần xác suất hậu nghiệm nhận diện khách hàng đắn đo ở ranh giới.
  • • Cụm lệch kích thước: Tự học trọng số trộn $\pi_k$ cho các cụm khác biệt quy mô.
